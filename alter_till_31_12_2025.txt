DROP PROCEDURE IF EXISTS split_and_insert;
DELIMITER $$

CREATE PROCEDURE split_and_insert()
BEGIN
  DECLARE done INT DEFAULT 0;
  DECLARE master_id INT;
  DECLARE sources TEXT;
  DECLARE app_id INT;
  DECLARE company_id INT;
  DECLARE dtype VARCHAR(100);
  DECLARE cur CURSOR FOR 
    SELECT id, data_sorce, a_application_login_id, company_masters_id, data_type 
    FROM custom_field_form_masters;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

  OPEN cur;
  read_loop: LOOP
    FETCH cur INTO master_id, sources, app_id, company_id, dtype;
    IF done THEN
      LEAVE read_loop;
    END IF;

    WHILE LOCATE(',', sources) > 0 DO
      INSERT INTO custom_field_form_datavalues (
        custom_field_master_id, data_sorce, a_application_login_id, company_masters_id, data_type
      )
      VALUES (
        master_id,
        TRIM(SUBSTRING_INDEX(sources, ',', 1)),
        app_id,
        company_id,
        dtype
      );
      SET sources = SUBSTRING(sources, LOCATE(',', sources) + 1);
    END WHILE;

    IF TRIM(sources) <> '' THEN
      INSERT INTO custom_field_form_datavalues (
        custom_field_master_id, data_sorce, a_application_login_id, company_masters_id, data_type
      )
      VALUES (
        master_id,
        TRIM(sources),
        app_id,
        company_id,
        dtype
      );
    END IF;

  END LOOP;
  CLOSE cur;
END$$
DELIMITER ;

CALL split_and_insert();
