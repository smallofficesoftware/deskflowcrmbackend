find unique key quary.

SELECT 
    TABLE_SCHEMA AS database_name,
    TABLE_NAME AS table_name,
    INDEX_NAME AS unique_index_name,
    GROUP_CONCAT(COLUMN_NAME ORDER BY SEQ_IN_INDEX) AS unique_columns
FROM 
    information_schema.STATISTICS
WHERE 
    NON_UNIQUE = 0  -- means UNIQUE index
    AND TABLE_SCHEMA = 'your_database_name'  -- replace with your DB name
GROUP BY 
    TABLE_SCHEMA, TABLE_NAME, INDEX_NAME
ORDER BY 
    TABLE_NAME, INDEX_NAME;
